cmake_minimum_required(VERSION 3.16)
project(riscv-pipeline CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB_RECURSE SOURCES "src/*.cpp")

add_executable(riscv-pipeline ${SOURCES})

target_include_directories(riscv-pipeline PRIVATE src)

add_executable(tests 
    tests/alu_tb.cpp
    tests/mux_tb.cpp
    tests/reg_file_tb.cpp
    tests/control_unit_tb.cpp
    tests/decoder_tb.cpp
    tests/memory_tb.cpp
    tests/cpu_tb.cpp

    src/datapath/alu.cpp
    src/datapath/mux.cpp
    src/datapath/reg_file.cpp
    src/control/control_unit.cpp
    src/control/decoder.cpp
    src/control/hazard_detection_unit.cpp
    src/memory/memory.cpp
    src/memory/memory_formatter.cpp
    src/cpu/cpu.cpp
)
target_include_directories(tests PRIVATE src external)
